```go
package daemon

import (
	"bytes"
	"context"
	"encoding/json"
	"fmt"
	"io"
	"net/http"
	"strings"

	"github.com/3toInf/hetu/internal/api"
)

// WebServer serves the HTTP JSON API + static frontend on top of the unix-socket
// RPC: each /api/* request is turned into an api.Request and dispatched through
// Server.dispatch, with an io.Writer adapter translating the newline-JSON reply
// into a standard HTTP response. This reuses every op's logic verbatim.
type WebServer struct {
	srv  *Server
	addr string // listen address ("127.0.0.1:19191"); "" = serve via caller's net/http
	// static dir to serve instead of the embedded dist ("" = embed)
	webDir string
}

func NewWebServer(srv *Server, addr, webDir string) *WebServer { return &WebServer{srv: srv, addr: addr, webDir: webDir} }

// httpWriter adapts the newline-JSON api.Encode stream (one api.Response per op)
// into HTTP: OK → 200 + raw Body; error → 4xx + {"error": ...}.
type httpWriter struct{ w http.ResponseWriter }

func (h *httpWriter) Write(p []byte) (int, error) {
	var resp api.Response
	if err := json.Unmarshal(bytes.TrimSpace(p), &resp); err != nil {
		http.Error(h.w, "bad server response", http.StatusInternalServerError)
		return len(p), nil
	}
	if !resp.OK {
		h.w.Header().Set("Content-Type", "application/json")
		http.Error(h.w, resp.Err, http.StatusBadRequest)
		return len(p), nil
	}
	h.w.Header().Set("Content-Type", "application/json")
	h.w.WriteHeader(http.StatusOK)
	_, _ = h.w.Write(resp.Body)
	return len(p), nil
}

func (ws *WebServer) call(ctx context.Context, w http.ResponseWriter, op string, body any) {
	var raw []byte
	if body != nil {
		raw, _ = json.Marshal(body)
	}
	ws.srv.dispatch(ctx, &httpWriter{w: w}, api.Request{Op: op, Body: raw})
}

func (ws *WebServer) Handler() http.Handler {
	mux := http.NewServeMux()
	mux.HandleFunc("GET /api/projects", func(w http.ResponseWriter, r *http.Request) {
		ws.call(r.Context(), w, "list_projects", api.ListProjectsReq{})
	})
	mux.HandleFunc("GET /api/sessions", func(w http.ResponseWriter, r *http.Request) {
		ws.call(r.Context(), w, "list_sessions", api.ListSessionsReq{
			ProjectPath: r.URL.Query().Get("project_path"),
			Status:      r.URL.Query().Get("status"),
			Agent:       r.URL.Query().Get("agent"),
		})
	})
	mux.HandleFunc("GET /api/sessions/{id}", func(w http.ResponseWriter, r *http.Request) {
		ws.call(r.Context(), w, "get_session", api.GetSessionReq{ID: r.PathValue("id")})
	})
	mux.HandleFunc("GET /api/search", func(w http.ResponseWriter, r *http.Request) {
		ws.call(r.Context(), w, "search", api.SearchReq{Q: r.URL.Query().Get("q")})
	})
	mux.HandleFunc("POST /api/discover", func(w http.ResponseWriter, r *http.Request) {
		ws.call(r.Context(), w, "discover", api.DiscoverReq{})
	})
	mux.HandleFunc("/", ws.staticHandler())
	return mux
}
```
（`staticHandler()` 在 Task 2 实现；Task 1 先返回 404 或 placeholder——为让本任务独立可测，先返回 `http.NotFound`，Task 2 替换。）

```go
// staticHandler serves the frontend: the embedded dist by default, a disk dir
// when webDir is set. Unknown non-API paths fall back to index.html (SPA routes).
func (ws *WebServer) staticHandler() http.Handler {
	var fsys http.FileSystem
	if ws.webDir != "" {
		fsys = http.Dir(ws.webDir)
	} else {
		fsys = http.FS(web.Dist)
	}
	fh := http.FileServer(fsys)
	return http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
		p := strings.TrimPrefix(r.URL.Path, "/")
		if p == "" { p = "index.html" }
		if f, err := fsys.Open(p); err == nil {
			f.Close()
			fh.ServeHTTP(w, r)
			return
		}
		// SPA fallback: serve index.html for any unknown non-API path
		if strings.HasPrefix(r.URL.Path, "/api/") {
			http.NotFound(w, r)
			return
		}
		r2 := new(http.Request)
		*r2 = *r
		r2.URL.Path = "/index.html"
		fh.ServeHTTP(w, r2)
	})
}
```
（import 加 `"github.com/3toInf/hetu/web"` 与 `"net/http"` 的 fs 用 `http.FS`；`webDir` 检查用 `os.Stat` 校验目录存在更稳——若 `--web-dir` 指定的目录不存在，启动时报错而非运行期 404，见 Task 3。）

## Self-Review（作者注）

**Spec 覆盖：** §4 端点 → T1；§4 静态 + §6 分发 → T2（embed/disk/SPA fallback）；§5 配置 → T3；§6 前端结构 → T4（脚手架）+ T5（视图）；§8 测试 → T1/T2 单测 + T6 e2e；§10 分发 → T2 embed + Makefile。

**设计要点确认：**
1. **go:embed 目录约束**：不能 `..` 跨包，所以 embed 放 `web/embed.go`（`//go:embed all:dist`），daemon `import web`。`web/dist` 必须含至少一个文件 → placeholder 先提交，Vite build 覆盖。
2. **httpWriter 适配器**：dispatch 每 op 恰写一条 `api.Response`（newline-JSON）；适配器解析后按 OK/err 转 HTTP。仅暴露非流式 op，无多写风险。
3. **staticHandler SPA fallback**：非 `/api/` 未知路径回退 index.html；`/api/*` 404。
4. **dist 必须提交**：`web/dist` 不能被 gitignore（M2 起它是 embed 输入）。`web/.gitignore` 只忽略 `node_modules`。

**类型一致性：** `api.SessionDTO`↔`Session`、`ProjectDTO`↔`Project`、`MessageDTO`↔`Message`、`SearchResultDTO`↔`SearchResult`（TS，snake_case 对齐）。

**风险：**
- `make build` 现在依赖 `npm`（`make web`）。无 node 环境退化：`go build` 仍可用 placeholder dist（界面显示"未构建"）。
- `npm install` 需网络；首次较慢。
- 前端无法在无浏览器环境端到端 UI 测试；M2 用 curl e2e + 构建通过代替。
- HTTP 默认开启改变了 hetud 默认行为（新增 localhost 端口）——`--no-web` 可关，文档注明。

# Hetu v0.2 — HITL + Notifications + Needs Attention Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Add Human-in-the-Loop tool approval, desktop notifications, and Needs Attention/Unread to Hetu's v0.1 Claude driver — so a user can see which sessions wait on them and approve/deny tool calls without leaving the terminal.

**Architecture:** v0.2 is an incremental layer on v0.1's raw-`claude`-CLI-subprocess driver. HITL works via a blocking `PermissionRequest` hook: claude spawns `hetu __hook-permission` on each permission decision; that client blocks on a daemon RPC (`ApprovalBroker`) until the user runs `hetu approve|deny` or answers inline in `hetu watch`. Status grows 5→7; store gains `last_viewed_at`; a `Notifier` fires on status transitions. No Agent SDK, no auto-approve policy, no defer/resume (all v0.3).

**Tech Stack:** Go 1.25, `modernc.org/sqlite` (pure-Go), cobra CLI, unix-socket JSON RPC, embedded SQL migrations, `notify-send`/`osascript` for desktop notifications.

## Global Constraints

- **Platform:** Linux + macOS only. unix-socket + pure-Go deps; cross-compile must stay one-line (`GOOS=darwin GOARCH=arm64 go build`). No CGO, no new runtime deps.
- **Hook cap:** PermissionRequest hook `timeout` = 540s (≤ 600s hard limit, 60s margin). On timeout the tool is denied.
- **HITL boundary:** Hetu does NO tool classification — it surfaces exactly what Claude's own permission logic raises (PermissionRequest fires only on tools Claude would prompt for). Read-only tools never reach Hetu.
- **No auto-approve:** v0.2 is manual approve/deny only. Allowlists/policy → v0.3.
- **Claude-only, Local-only** this milestone.
- **TDD:** every code task writes the failing test first; `go test ./...` and `go vet ./...` must stay green after every task. Commit per task.
- **Conventions:** conventional-commit messages (`feat(scope):`, `fix(scope):`, `refactor(scope):`, `docs:`); follow existing file/fn naming; keep real claude deps isolated in `internal/claude`.
- **Wire format:** daemon↔client is newline-delimited JSON (`api.Request{Op,Body}` / `api.Response{OK,Err,Body}`) over unix socket, via `api.Encode`.

**Spec:** [`docs/superpowers/specs/2026-08-14-hetu-v0.2-design.md`](../specs/2026-08-14-hetu-v0.2-design.md). Read it before starting.

```go
func (m *SessionManager) RequestApproval(ctx context.Context, hetuID, toolUseID, toolName, toolInput string) (ApprovalDecision, error) {
	m.mu.Lock()
	l, ok := m.live[hetuID]
	m.mu.Unlock()
	if !ok {
		return ApprovalDecision{}, errNotDriven
	}
	entry := &pendingEntry{
		PendingApproval: PendingApproval{ToolUseID: toolUseID, ToolName: toolName, ToolInput: toolInput},
		ch:              make(chan ApprovalDecision, 1),
	}
	l.mu.Lock()
	if l.pending == nil {
		l.pending = map[string]*pendingEntry{}
	}
	l.pending[toolUseID] = entry
	l.mu.Unlock()

	l.broadcast(agent.Event{Type: agent.EventApproval, ToolName: toolName, ToolUseID: toolUseID,
		ToolJSON: toolInput, ApprovalState: "pending"})
	m.setLiveStatus(l, hetuID, session.StatusWaitingForApproval)

	select {
	case d := <-entry.ch:
		state := "allowed"
		if !d.Allow {
			state = "denied"
		}
		l.broadcast(agent.Event{Type: agent.EventApproval, ToolUseID: toolUseID, ApprovalState: state})
		m.setLiveStatus(l, hetuID, session.StatusRunning)
		return d, nil
	case <-ctx.Done():
		l.mu.Lock()
		delete(l.pending, toolUseID)
		l.mu.Unlock()
		l.broadcast(agent.Event{Type: agent.EventApproval, ToolUseID: toolUseID, ApprovalState: "timeout"})
		m.setLiveStatus(l, hetuID, session.StatusRunning)
		return ApprovalDecision{Allow: false, Reason: "approval timed out"}, ctx.Err()
	}
}

func (m *SessionManager) ResolveApproval(hetuID, toolUseID string, d ApprovalDecision) bool {
	m.mu.Lock()
	l, ok := m.live[hetuID]
	m.mu.Unlock()
	if !ok {
		return false
	}
	l.mu.Lock()
	entry, found := l.pending[toolUseID]
	if found {
		delete(l.pending, toolUseID)
	}
	l.mu.Unlock()
	if !found {
		return false
	}
	entry.ch <- d
	return true
}

func (m *SessionManager) PendingApprovals(hetuID string) []PendingApproval {
	m.mu.Lock()
	l, ok := m.live[hetuID]
	m.mu.Unlock()
	if !ok {
		return nil
	}
	l.mu.Lock()
	defer l.mu.Unlock()
	out := make([]PendingApproval, 0, len(l.pending))
	for _, e := range l.pending {
		out = append(out, e.PendingApproval)
	}
	return out
}
```

```go
func newWatchCmd() *cobra.Command {

		RunE: func(cmd *cobra.Command, args []string) error {
			ctx := cmd.Context()
			ch, err := newClient().Watch(ctx, args[0])
			if err != nil { return err }
			in := bufio.NewReader(os.Stdin)
			for ev := range ch {

					if err := newClient().Approve(ctx, args[0], ev.ToolUseID, allow, ""); err != nil {
						return err
					}
				}
			}
			return nil
```
